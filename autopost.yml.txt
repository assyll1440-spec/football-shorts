name: autopost

on:
  schedule:
    - cron: "17 3 * * *"   # every day at 03:17 UTC (08:17 Almaty)
  workflow_dispatch:        # "Run workflow" button for manual runs

permissions:
  contents: write

env:
  VOICE: am_puck        # Kokoro voice. Leave empty ('') for text-only videos
  SPEED: "1.05"
  ESPEAK_BIN: /usr/bin/espeak-ng

jobs:
  render:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4

      - name: Arrange files into folders
        run: |
          mkdir -p fonts queue done
          for f in *.ttf; do [ -e "$f" ] && git mv "$f" fonts/; done
          for f in f[0-9]*.json; do [ -e "$f" ] && git mv "$f" queue/; done
          true

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Install tools
        run: |
          sudo apt-get update -q
          sudo apt-get install -y -q ffmpeg espeak-ng
          pip install -q numpy==1.26.4 onnxruntime==1.19.2 playwright==1.47.0
          python -m playwright install --with-deps chromium

      - name: Download voice model (Kokoro)
        if: env.VOICE != ''
        run: |
          mkdir -p models
          for f in kokoro-v1.0.onnx voices-v1.0.bin; do
            curl -sSL -o models/$f https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.0/$f
          done

      - name: Make today's video
        id: make
        run: python run.py

      - name: Publish video as a release
        if: steps.make.outputs.slug != ''
        env:
          GH_TOKEN: ${{ github.token }}
          SLUG: ${{ steps.make.outputs.slug }}
          TITLE: ${{ steps.make.outputs.title }}
        run: |
          gh release create "$SLUG" "output/$SLUG.mp4" "output/$SLUG.txt" \
            --title "$TITLE" --notes-file "output/$SLUG.txt"

      - name: Save queue progress
        if: steps.make.outputs.slug != ''
        run: |
          git config user.name "autopost-bot"
          git config user.email "autopost-bot@users.noreply.github.com"
          git add -A fonts queue done && git add -u .
          git commit -m "Rendered ${{ steps.make.outputs.slug }} (${{ steps.make.outputs.left }} left)"
          git push
